cmake_minimum_required(VERSION 3.24)
project(vega-qt-assistant-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Network Test)
enable_testing()
add_executable(assistant-test assistant_test.cpp ../src/assistantbackend.cpp ../src/assistantbackend.h)
target_include_directories(assistant-test PRIVATE ../src)
target_link_libraries(assistant-test PRIVATE Qt6::Core Qt6::Network Qt6::Test)
target_compile_options(assistant-test PRIVATE -Wall -Wextra -Werror)
add_test(NAME assistant-request-lifecycle COMMAND assistant-test)
set_tests_properties(assistant-request-lifecycle PROPERTIES TIMEOUT 30)
